using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Reflection;
using Blacksmith;
using UnityEditor;
using UnityEditor.SceneManagement;

using UnityEngine.InputSystem.LowLevel;
using UnityEngine.Rendering;

[InitializeOnLoad]
public static class CampaignValidate
{
    [Serializable]public sealed class Check{public string name;public bool passed;}
    [Serializable]public sealed class Result{public string runId,status,projectPath,unityVersion,persistentDataPath,message;public bool isPlaying;public List<Check> checks=new();public List<string> errors=new();}
    static Result result=new();static string marker,runId;static double deadline,next;static int step;static float startX;static bool active;
    static CampaignController C=>CampaignController.Instance;static SmithingLoop L=>SmithingLoop.Instance;
    static CampaignValidate(){EditorApplication.playModeStateChanged+=ModeChanged;}
    public static void Run()
    {
        next=EditorApplication.timeSinceStartup+8;EditorApplication.update+=WaitForEditor;
    }
    static void WaitForEditor()
    {
        if(EditorApplication.isCompiling||EditorApplication.isUpdating)next=EditorApplication.timeSinceStartup+8;
        if(EditorApplication.timeSinceStartup<next)return;
        EditorApplication.update-=WaitForEditor;Begin();
    }
    static void Begin()
    {
        if(!Application.isBatchMode||PlayerSettings.companyName!="BatterMapValidation")throw new Exception("isolated validation only");
        var args=Environment.GetCommandLineArgs();string Arg(string key,string fallback){int n=Array.IndexOf(args,key);return n>=0&&n+1<args.Length?args[n+1]:fallback;}
        marker=Arg("-validationMarker",Path.Combine(Directory.GetParent(Application.dataPath).FullName,"campaign-validation.json"));runId=Arg("-validationRunId","c008-stage");
        SessionState.SetString("C008Marker",marker);SessionState.SetString("C008Run",runId);SessionState.SetBool("C008Active",true);
        // The supplied validation identity is unique; reset only its test fixture file.
        string save=Path.Combine(Application.persistentDataPath,"smithing-loop-v1.json");if(File.Exists(save))File.Move(save,save+".fixture-"+Guid.NewGuid().ToString("N"));
        try{}catch(Exception e){result.errors.Add(e.ToString());Finish(false,e.Message);return;}SessionState.SetString("C008Domain",JsonUtility.ToJson(result));
        EditorSceneManager.OpenScene("Assets/Scenes/SampleScene.unity",OpenSceneMode.Single);EditorApplication.isPlaying=true;
    }
    static void Assert(bool condition,string name){result.checks.Add(new Check{name=name,passed=condition});if(!condition)throw new Exception(name);}
    static void ModeChanged(PlayModeStateChange change)
    {
        if(change!=PlayModeStateChange.EnteredPlayMode||!SessionState.GetBool("C008Active",false))return;
        marker=SessionState.GetString("C008Marker","");runId=SessionState.GetString("C008Run","");result=JsonUtility.FromJson<Result>(SessionState.GetString("C008Domain","{}"));
        step=0;deadline=EditorApplication.timeSinceStartup+150;next=EditorApplication.timeSinceStartup+2;active=true;Application.logMessageReceived+=Log;EditorApplication.update+=Tick;
    }
    static void Log(string message,string trace,LogType type){if(type==LogType.Error||type==LogType.Exception||type==LogType.Assert)result.errors.Add(message);}
    static bool opaque, fading;
    static Vector3 cancelledPosition;static int portalHealth;
    static void PressF(){InputSystem.QueueStateEvent(Keyboard.current,new KeyboardState(Key.F));InputSystem.Update();}
    static void Release(){InputSystem.QueueStateEvent(Keyboard.current,new KeyboardState());InputSystem.Update();}
    static CampaignWorldObject Obj(string name)=>GameObject.Find(name).GetComponent<CampaignWorldObject>();
    static void Near(string name)
    {
        var box=Obj(name).GetComponent<BoxCollider2D>();
        C.Teleport(new Vector2(box.bounds.center.x,box.bounds.min.y+1.1f));
        Release();
    }
    static bool HasButton(string name)=>Object.FindObjectsByType<Button>(FindObjectsSortMode.None).Any(x=>x.name==name);
    static void Tick()
    {
        if(!active)return;
        if(C!=null&&C.Travelling){float a=C.GetComponent<CampaignTransition>().Alpha;opaque|=a>=.99f;fading|=a>0&&a<1;}
        if(EditorApplication.timeSinceStartup<next)return;
        try
        {
            if(EditorApplication.timeSinceStartup>deadline)throw new Exception("C011 timeout, step "+step);
            if(C==null||!C.Ready)return;
            switch(step)
            {
                case 0:
                    Assert(SmithyInterior.Instance!=null,"saved smithy prefab spawns");
                    Assert(SmithyInterior.Instance.entry!=null&&SmithyInterior.Instance.exit!=null&&SmithyInterior.Instance.cameraBounds!=null,"prefab has entry exit and camera references");
                    Assert(Object.FindObjectsByType<SmithyStation>(FindObjectsSortMode.None).Length==4,"four physical quick interaction stations exist");
                    Assert(C.State.toolsLinked&&L.SmithData.equipment.Count(x=>x.slot=="Pickaxe"||x.slot=="Axe")==2,"legacy tool ownership migrates once");
                    if(Keyboard.current==null)InputSystem.AddDevice<Keyboard>();if(Mouse.current==null)InputSystem.AddDevice<Mouse>();
                    InputSystem.settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;
                    InputSystem.settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;
                    InputSystem.settings.updateMode=InputSettings.UpdateMode.ProcessEventsManually;
                    var pi=C.Player.GetComponent<PlayerInput>();pi.enabled=false;pi.enabled=true;pi.SwitchCurrentControlScheme(Keyboard.current,Mouse.current);pi.ActivateInput();
                    var handler=C.Player.GetComponent<PlayerInputHandler>();handler.enabled=false;handler.enabled=true;
                    GameUIController.Instance.Play();C.Teleport(C.townSpawn);C.State.gold=1000;C.UI.ShowDialogue("store");Click("BrowseGoods");Click("Goods_pick_2");Click("Purchase");
                    Assert(L.SmithData.bag.Any(x=>x.itemId=="town_pick_2"&&x.count==1)&&Tool(ToolType.Pickaxe)?.Tier==1,"shop purchase creates owned tool without auto equip");C.UI.Close();C.UI.Npc.Advance();C.UI.Npc.Advance();Near("BlackSmith");step++;break;
                case 1:PressF();step++;break;
                case 2:
                    Release();if(C.Travelling)return;
                    Assert(SmithyInterior.Instance.Inside&&C.InTown&&!L.InShop,"town BlackSmith F enters physical interior");
                    Assert(opaque&&fading,"smithy entrance has partial and opaque fade");
                    startX=C.Player.position.x;InputSystem.QueueStateEvent(Keyboard.current,new KeyboardState(Key.D));InputSystem.Update();step++;next=EditorApplication.timeSinceStartup+1;return;
                case 3:
                    Release();Assert(C.Player.position.x>startX+1&&C.Player.position.y>61&&C.Player.position.y<65,"real movement traverses interior collision floor");Assert(SmithyInterior.Instance.GetComponentsInChildren<Rigidbody2D>().All(x=>!x.simulated),"decorative props have no active physics");
                    Assert(SmithyInterior.Instance.GetComponentsInChildren<Rigidbody2D>().All(x=>x.position.y>=61),"decorative props stay on the indoor floor");Capture("c011-interior.png");
                    var cam=Camera.main;var pos=cam.transform.position;float zoom=cam.orthographicSize;cam.transform.position=new Vector3(-252,67,pos.z);cam.orthographicSize=16;Capture("c011-interior-overview.png");cam.transform.position=pos;cam.orthographicSize=zoom;
                    NearStation(SmithyStationKind.Rack);step++;break;
                case 4:PressF();step++;break;
                case 5:
                    Release();if(!ReadyPanel())return;
                    Assert(S.State==ScreenState.Chest&&Object.FindObjectsByType<InventorySlotView>(FindObjectsSortMode.None).Any(x=>x.Stack.itemId=="town_pick_1")&&Object.FindObjectsByType<InventorySlotView>(FindObjectsSortMode.None).Any(x=>x.Stack.itemId=="town_axe_1"),"rack F opens equipment with gathering tool slots");
                    InventoryService.Add(S.Inventory.Data.chest,new Blacksmith.Stack("town_pick_2",1));InventoryService.Add(S.Inventory.Data.chest,new Blacksmith.Stack("town_axe_2",1));
                    Assert(S.Inventory.Equip(S.Inventory.Data.chest.First(x=>x.itemId=="town_pick_2"))&&S.Inventory.Equip(S.Inventory.Data.chest.First(x=>x.itemId=="town_axe_2")),"rack equips both upgraded tools and returns previous items");
                    Debug.Log("IconDiagnostics content pick="+Resources.Load<SmithingLoopContent>("SmithingLoopContent").pickaxeIcon+" axe="+Resources.Load<SmithingLoopContent>("SmithingLoopContent").axeIcon+" resolved="+L.ToolIcon("linked_pick"));Assert(S.view.Art("linked_pick")!=null&&S.view.Art("linked_axe")!=null,"gathering tool icons resolve in equipment UI");
                    S.OpenEquipmentRack();L.SendMessage("Update");Capture("c011-rack.png");S.SetState(ScreenState.Home);step++;break;
                case 6:
                    if(L.InShop||L.Transitioning)return;
                    Assert(Tool(ToolType.Pickaxe)?.Tier==2&&Tool(ToolType.Axe)?.Tier==2,"rack tools apply to real gathering slots");
                    var stack=new Blacksmith.Stack("wood",3,Quality.Finest);Assert(C.Inventory.TryAddItem(L.FieldItemFor(stack),3),"quality stack can be carried in field");NearStation(SmithyStationKind.Chest);step++;break;
                case 7:PressF();step++;break;
                case 8:
                    Release();if(!ReadyPanel())return;
                    Assert(S.State==ScreenState.Chest&&!HasButton("Equip_Weapon"),"chest F opens bag and storage view");
                    var quality=S.Inventory.Data.bag.Single(x=>x.itemId=="wood"&&x.quality==Quality.Finest);
                    Assert(quality.count==3&&!C.Inventory.Items.Any(x=>x.itemData.ItemId=="smith:wood:3"),"field ownership moves into crafting bag preserving quality");
                    Assert(S.Inventory.Transfer(S.Inventory.Data.bag,S.Inventory.Data.chest,quality,3),"bag stack transfers to crafting chest");
                    quality=S.Inventory.Data.chest.Single(x=>x.itemId=="wood"&&x.quality==Quality.Finest);
                    Assert(S.Inventory.Transfer(S.Inventory.Data.chest,S.Inventory.Data.bag,quality,3),"chest stack returns to bag");
                    string gear=S.catalog.items.First(x=>x.equipmentSlot=="Weapon"&&!x.bow&&!x.twoHanded).id;
                    InventoryService.Add(S.Inventory.Data.bag,new Blacksmith.Stack(gear,2,Quality.Finest));
                    Assert(S.Inventory.Equip(S.Inventory.Data.bag.First(x=>x.itemId==gear)),"crafted equipment can equip directly from shared bag");
                    S.SetState(ScreenState.Chest);L.SendMessage("Update");Capture("c011-chest.png");S.SetState(ScreenState.Home);step++;break;
                case 9:
                    if(L.InShop||L.Transitioning)return;
                    Assert(C.Inventory.Items.Any(x=>x.itemData.ItemId=="smith:wood:3"&&x.quantity==3),"closing chest exports quality stack exactly once");
                    Assert(C.Inventory.Items.Any(x=>x.itemData.ItemId.StartsWith("smith:")&&x.itemData.ItemId!="smith:wood:3"),"crafted equipment is carried in field inventory");
                    Assert(!L.SmithData.bag.Any(),"successful export leaves no duplicate smith bag stacks");
                    Assert(L.SaveSlot(1),"manual save accepts bridged items");
                    var weapon=C.Player.GetComponent<PlayerToolController>();Assert(Enumerable.Range(0,weapon.ToolSlotCount).Select(weapon.GetToolAtSlot).Any(t=>t!=null&&t.IsWeapon&&t.Damage>10),"crafted weapon quality affects field damage");
                    NearStation(SmithyStationKind.Bed);step++;break;
                case 10:PressF();step++;break;
                case 11:
                    Release();if(!ReadyPanel())return;
                    Assert(S.State==ScreenState.Sleep&&HasButton("SleepYes")&&HasButton("SleepNo"),"bed F opens confirm and cancel");L.SendMessage("Update");Capture("c011-bed.png");Click("SleepNo");step++;break;
                case 12:
                    if(L.InShop||L.Transitioning)return;Assert(!L.SmithData.night,"bed cancellation preserves day");PressF();step++;break;
                case 13:
                    Release();if(!ReadyPanel())return;Click("SleepYes");step++;break;
                case 14:
                    if(L.InShop||L.Transitioning)return;
                    Assert(L.SmithData.night&&L.SmithData.day==1&&C.Player.GetComponent<PlayerAssimilate>().CurrentAssimilation==C.Player.GetComponent<PlayerAssimilate>().MaxAssimilation,"day nap returns to physical room at night and heals");NearStation(SmithyStationKind.CraftingDoor);step++;break;
                case 15:PressF();step++;break;
                case 16:
                    Release();if(!ReadyPanel())return;
                    Assert(S.State==ScreenState.Workshop&&HasButton("Facility_5"),"crafting door F opens five station workshop");L.SendMessage("Update");Capture("c011-workshop.png");S.SetState(ScreenState.Home);step++;break;
                case 17:
                    if(L.InShop||L.Transitioning)return;NearStation(SmithyStationKind.Rack);step++;break;
                case 18:PressF();step++;break;
                case 19:
                    Release();if(!ReadyPanel())return;
                    S.Inventory.Unequip("Pickaxe");S.SetState(ScreenState.Home);step++;break;
                case 20:
                    if(L.InShop||L.Transitioning)return;
                    Assert(Tool(ToolType.Pickaxe)==null,"unequipping tool removes field tool rather than respawning it");
                    Assert(L.SaveSlot(2),"unequipped tools save without legacy migration repetition");NearStation(SmithyStationKind.Bed);step++;break;
                case 21:PressF();step++;break;
                case 22:Release();if(!ReadyPanel())return;Click("SleepYes");step++;break;
                case 23:
                    if(L.InShop||L.Transitioning)return;
                    Assert(!L.SmithData.night&&L.SmithData.day==2,"night sleep advances one morning and returns to room");
                    C.Teleport(SmithyInterior.Instance.exit.position+Vector3.right*.5f);step++;break;
                case 24:
                    if(C.Travelling||SmithyInterior.Instance.Inside)return;
                    Assert(C.InTown&&!HasButton("ConfirmYes")&&Vector2.Distance(C.Player.position,C.townScene.blackSmith.position)<4,"left doorway automatically fades back to town without confirmation");
                    Assert(L.LoadSlot(2),"bridged item save accepted for scene reload");step++;next=EditorApplication.timeSinceStartup+2;return;
                case 25:
                    if(!C.Ready||!L.Initialized)return;
                    Assert(SmithyInterior.Instance.Inside&&C.State.toolsLinked&&Tool(ToolType.Pickaxe)==null,"reload restores physical room and does not respawn unequipped tool");
                    Assert(C.Inventory.Items.Any(x=>x.itemData.ItemId=="smith:wood:3"&&x.quantity==3),"reload restores bridged quality item without loss");
                    Assert(L.SmithData.chest.Count(x=>x.itemId=="town_pick_2")==1,"reload preserves unequipped tool ownership once");
                    Assert(result.errors.Count==0,"physical smithy and shared inventory completed");Finish(true,"C011 physical stations, movement, tool equip, shared items and scene reload verified");return;
            }
            next=EditorApplication.timeSinceStartup+.25;
        }catch(Exception e){result.errors.Add(e.ToString());Finish(false,e.Message);}
    }
    static BlacksmithController S=>Object.FindFirstObjectByType<BlacksmithController>();
    static bool ReadyPanel()=>L.InShop&&!L.Transitioning&&S!=null&&S.Inventory!=null;
    static ToolData Tool(ToolType type){var t=C.Player.GetComponent<PlayerToolController>();return Enumerable.Range(0,t.ToolSlotCount).Select(t.GetToolAtSlot).FirstOrDefault(x=>x!=null&&x.ToolType==type);}
    static void NearStation(SmithyStationKind kind){var t=Object.FindObjectsByType<SmithyStation>(FindObjectsSortMode.None).Single(x=>x.kind==kind);C.Teleport(t.transform.position+Vector3.up*1.2f);Release();}
    static void Click(string name){var button=Object.FindObjectsByType<Button>(FindObjectsSortMode.None).First(x=>x.name==name);button.onClick.Invoke();}
    static void Capture(string filename)
    {
        var camera=Camera.main;var canvases=Object.FindObjectsByType<Canvas>(FindObjectsSortMode.None).Where(c=>c.renderMode==RenderMode.ScreenSpaceOverlay).ToArray();
        var orders=canvases.ToDictionary(c=>c,c=>(c.sortingLayerID,c.sortingOrder));
        var topLayer=SortingLayer.layers.OrderBy(l=>l.value).Last().id;
        var texture=new Texture2D(1920,1080,TextureFormat.RGB24,false);var target=new RenderTexture(1920,1080,24);var previous=RenderTexture.active;var oldTarget=camera.targetTexture;float oldAspect=camera.aspect;
        try
        {
            camera.targetTexture=target;camera.aspect=1920f/1080;
            foreach(var canvas in canvases){canvas.renderMode=RenderMode.ScreenSpaceCamera;canvas.worldCamera=camera;canvas.planeDistance=1;canvas.sortingLayerID=topLayer;canvas.sortingOrder=20000+orders[canvas].sortingOrder;}
            Canvas.ForceUpdateCanvases();RenderPipeline.SubmitRenderRequest(camera,new RenderPipeline.StandardRequest{destination=target});RenderTexture.active=target;texture.ReadPixels(new Rect(0,0,1920,1080),0,0);texture.Apply();File.WriteAllBytes(Path.Combine(Path.GetDirectoryName(marker),filename),texture.EncodeToPNG());
        }
        finally{foreach(var canvas in canvases){canvas.renderMode=RenderMode.ScreenSpaceOverlay;canvas.worldCamera=null;canvas.sortingLayerID=orders[canvas].sortingLayerID;canvas.sortingOrder=orders[canvas].sortingOrder;}camera.targetTexture=oldTarget;camera.aspect=oldAspect;RenderTexture.active=previous;Object.DestroyImmediate(texture);Object.DestroyImmediate(target);}
    }
    static void Finish(bool passed,string message)
    {
        active=false;EditorApplication.update-=Tick;Application.logMessageReceived-=Log;SessionState.SetBool("C008Active",false);
        result.runId=runId;result.status=passed?"passed":"failed";result.projectPath=Directory.GetParent(Application.dataPath).FullName;result.unityVersion=Application.unityVersion;result.persistentDataPath=Application.persistentDataPath;result.isPlaying=Application.isPlaying;result.message=message;
        File.WriteAllText(marker,JsonUtility.ToJson(result,true));Debug.Log("C008_VALIDATION_"+(passed?"PASSED":"FAILED"));Debug.Log("BATTERMAP_RUNTIME_COMPLETE "+runId);EditorApplication.Exit(passed?0:1);
    }
}
